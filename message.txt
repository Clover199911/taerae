import React, { useState } from 'react';
import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card';

const EarningsTable = () => {
  const [selectedRarity, setSelectedRarity] = useState('Mythic');
  const [selectedCondition, setSelectedCondition] = useState('Pristine');
  
  const earnings = [
    {
      currency: "Crystals",
      ranges: {
        "Standard": { min: 50, max: 100 },
        "Unique": { min: 250, max: 500 },
        "Glyph": { min: 500, max: 1000 },
        "Mythic": { min: 1500, max: 2500 }
      }
    },
    {
      currency: "Fantasia Tokens",
      ranges: {
        "Unique": { min: 1, max: 5 },
        "Glyph": { min: 5, max: 10 },
        "Mythic": { min: 8, max: 15 }
      }
    },
    {
      currency: "Stardust",
      ranges: {
        "Unique": { min: 0, max: 3 },
        "Glyph": { min: 3, max: 7 },
        "Mythic": { min: 10, max: 15 }
      }
    },
    {
      currency: "Reverie Gems",
      ranges: {
        "Glyph": { min: 5, max: 10 },
        "Mythic": { min: 10, max: 15 }
      }
    },
    {
      currency: "Astral Essence",
      ranges: {
        "Mythic": { min: 10, max: 20 }
      }
    }
  ];

  const conditions = {
    "Pristine": 1.5,
    "Mint": 1.25,
    "Good": 1,
    "Worn": 0.75,
    "Damaged": 0.5
  };

  const rarities = ["Mythic", "Glyph", "Unique", "Standard"];
  const conditionsList = Object.keys(conditions);

  const calculateRange = (min, max, condition) => {
    const multiplier = conditions[condition];
    return {
      min: Math.floor(min * multiplier),
      max: Math.floor(max * multiplier)
    };
  };

  const calculate10HourRange = (range) => {
    return {
      min: range.min * 30, // 30 runs in 10 hours
      max: range.max * 30
    };
  };

  return (
    <Card className="w-full">
      <CardHeader>
        <CardTitle>Dreamwalk 10-Hour Earnings Range</CardTitle>
      </CardHeader>
      <CardContent>
        <div className="mb-4 space-x-2">
          <select 
            value={selectedRarity} 
            onChange={(e) => setSelectedRarity(e.target.value)}
            className="border rounded p-1"
          >
            {rarities.map(rarity => (
              <option key={rarity} value={rarity}>{rarity}</option>
            ))}
          </select>
          <select 
            value={selectedCondition} 
            onChange={(e) => setSelectedCondition(e.target.value)}
            className="border rounded p-1"
          >
            {conditionsList.map(condition => (
              <option key={condition} value={condition}>{condition}</option>
            ))}
          </select>
        </div>
        <div className="overflow-x-auto">
          <table className="w-full">
            <thead>
              <tr className="bg-gray-100">
                <th className="p-2 text-left">Currency</th>
                <th className="p-2 text-right">Per Run Range</th>
                <th className="p-2 text-right">10-Hour Total Range</th>
              </tr>
            </thead>
            <tbody>
              {earnings.map(currency => {
                if (!currency.ranges[selectedRarity]) return null;
                
                const perRunRange = calculateRange(
                  currency.ranges[selectedRarity].min,
                  currency.ranges[selectedRarity].max,
                  selectedCondition
                );
                
                const totalRange = calculate10HourRange(perRunRange);
                
                return (
                  <tr key={currency.currency} className="border-t">
                    <td className="p-2">{currency.currency}</td>
                    <td className="p-2 text-right">
                      {perRunRange.min.toLocaleString()} - {perRunRange.max.toLocaleString()}
                    </td>
                    <td className="p-2 text-right">
                      {totalRange.min.toLocaleString()} - {totalRange.max.toLocaleString()}
                    </td>
                  </tr>
                );
              })}
            </tbody>
          </table>
        </div>
        <div className="mt-4 text-sm text-gray-600">
          * Based on 30 runs over 10 hours (20-minute cooldown)
          <br/>
          * Actual earnings will be randomly distributed within these ranges
        </div>
      </CardContent>
    </Card>
  );
};

export default EarningsTable;
